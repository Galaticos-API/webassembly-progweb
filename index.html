(module
  ;; Aloca e exporta 10 páginas de 64 KB (640 KB total)
  (memory (export "memory") 10)

  ;; Função soma: recebe 2 inteiros de 32 bits e retorna a soma
  (func $add (export "add") (param $a i32) (param $b i32) (result i32)
    local.get $a
    local.get $b
    i32.add
  )

  ;; Fibonacci recursivo
  (func $fibonacci (export "fibonacci") (param $n i32) (result i32)
    local.get $n
    i32.const 2
    i32.lt_s
    if (result i32)
      local.get $n
    else
      local.get $n
      i32.const 1
      i32.sub
      call $fibonacci
      local.get $n
      i32.const 2
      i32.sub
      call $fibonacci
      i32.add
    end
  )
)
#define EMSCRIPTEN_KEEPALIVE __attribute__((used))

EMSCRIPTEN_KEEPALIVE
int add(int a, int b) {
    return a + b;
}

EMSCRIPTEN_KEEPALIVE
int fibonacci(int n) {
    if (n < 2) return n;
    return fibonacci(n - 1) + fibonacci(n - 2);
}

EMSCRIPTEN_KEEPALIVE
void applyGrayscale(unsigned char* pixels, int length) {
    for (int i = 0; i < length; i += 4) {
        unsigned char r = pixels[i];
        unsigned char g = pixels[i + 1];
        unsigned char b = pixels[i + 2];
        unsigned char gray = (unsigned char)((r * 299 + g * 587 + b * 114) / 1000);
        pixels[i] = gray;
        pixels[i + 1] = gray;
        pixels[i + 2] = gray;
    }
}

Estrutura em Pilha

O WebAssembly opera sobre uma máquina de pilha (stack machine). Instruções como local.get empilham valores e instruções como i32.add desempilham os operandos e empilham o resultado.

Memória Linear

A memória do Wasm é um vetor contíguo de bytes não tipados exposto ao JavaScript através de um ArrayBuffer. Isso permite que C/Rust use ponteiros normalmente.

Segurança e Sandbox

O código Wasm roda dentro do mesmo modelo de segurança restrito do JavaScript no navegador, sem acesso direto a arquivos locais ou chamadas de sistema não autorizadas.
